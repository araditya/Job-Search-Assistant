## Troubleshooting

# If backend fails to start:
# 1. Check if port 5000 is in use:
netstat -ano | findstr :5000

# 2. If in use, kill the process (replace <PID>):
taskkill /PID <PID> /F

# 3. Check for errors in error.log:
Get-Content .\error.log -Tail 100

# If frontend fails to start:
# 1. Check if port 3000 is in use:
netstat -ano | findstr :3000

# 2. Delete node_modules and reinstall:
cd frontend
Remove-Item -Recurse -Force node_modules
npm install
npm start

# If /search_jobs returns 500 error:
# 1. Check error.log in backend folder:
Get-Content error.log -Tail 200

# 2. Verify RAPIDAPI_KEY is set:
$env:RAPIDAPI_KEY

# 3. Check if API key is valid by testing manually:
curl -X GET "https://jsearch.p.rapidapi.com/search?query=python&page=1" `
  -H "X-RapidAPI-Key: 90a54da8d4msh83e07f3e5def448p1d8c5ejsn90e2d9bf3155" `
  -H "X-RapidAPI-Host: jsearch.p.rapidapi.com"
